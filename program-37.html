#include <lpc214x.h>
unsigned long int c=0xff;
__irq void T0_ISR (void)
{   c-=1;
    IO0PIN =  c<<16;    /* up count*/
    T0IR = ( T0IR | (0x01) );
    VICVectAddr = 0x00;
}
int main (void)
{
VPBDIV = 2; /* For Pclk = 30MHz */
IO0DIR = ( IO0DIR | (0X00ff0000) ); /* p1.25 as output pin for LED */
IO0PIN = IO0PIN | 0X00ff0000; /* Writing 1 to LED pin P1.25 */
VICVectAddr0 = (unsigned) T0_ISR; /* T0 ISR Address */
VICVectCntl0 = 0x00000024; /* Enable T0 IRQ slot */
VICIntEnable = 0x00000010; /* Enable T0 interrupt */
VICIntSelect = 0x00000000; /* T0 configured as IRQ */
T0TCR = 0x02; /* Reset TC and PR */
T0CTCR = 0x00; /* Timer mode, increment on every rising edge */
T0PR = 0x1D; /* Load Pre-Scalar counter with 29, timer to count every 1usec */
T0MR0 = 1000000; /* Load timer counter for 1sec delay */
T0MCR = 0x0003; /* Interrupt generate on match and reset timer */
T0TCR = 0x01; /* Enable timer */
while (1);
}
