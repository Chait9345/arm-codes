 INCLUDE VICVPB.S
 AREA INTRUPT, CODE,READONLY
 EXPORT __main  ;refer the line no 416 of startup.s file
__main
    ENTRY
;   LDR R1,=MEMMAP; User Flash Mode. Interrupt vectors are
;   LDR R0,=0X01  ;not re-mapped and reside in Flash.
;   STR R0,[R1]
    LDR R1,=IO1DIR    ; to configure P1.25 as output
    LDR R0,=0X02000000
    STR R0,[R1]
    LDR R1,=PINSEL0
    LDR R0,=0XC0; TO CONFIGURE P0.3 AS EINT1
    STR R0,[R1]
    LDR R1,=EXTMODE
    LDR R0,=0X02; TO SELECT EINT1 AS EDGE TRIGGERED(1 FOR EDGE, 0 FOR LEVEL)
    STR R0,[R1]
    LDR R1,=EXTPOLAR
    LDR R0,=0X0; TO SELECT EINT1 AS negative EDGE TRIGGERED(0 for negative, 1 for positive)
    STR R0,[R1]
    LDR R1,=VICVectCntl0; slot0 with highest priority
    LDR R0,=0x2F; EINT1 number is 15 and 2  for vectored IRQ slot is enabled
    STR R0,[R1]
    LDR R1,=VICVectAddr0
    LDR R0,=IRQ_EInt1; ISR address to be stored in VICVectAddr0
    STR R0,[R1]
    LDR R1,=VICIntEnable
    LDR R0,=0x00008000; enable EINT1
    STR R0,[R1]
S   B S
IRQ_EInt1
    STMDB R13!,{R0-R1}
    LDR R1,=EXTINT
    LDR R0,=0x2
    STR R0,[R1]
    ldr r2,=0x40000000; Starting address of data memory
    ldr r0,[r2]
    LDR R1,=IO1PIN
    EOR R0,R0,#0X02000000
    STR R0,[R1]
    str r0,[r2]
    LDR R1,=VICVectAddr
    LDR R0,=0
    STR R0,[R1]
    LDMIA R13!,{R0-R1}
    subs pc,r14,#4 ; Adjust the PC to jump back to where it was interrupted
    END
