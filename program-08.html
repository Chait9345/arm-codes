Program to find square of a decimal number (0 to 10) using lookup table

    AREA  SQUARE , CODE, READONLY
ENTRY                    ;Mark first instruction to execute
START
    LDR R0,=TABLE1       ; Load start address of Lookup table
    LDR R1,=INP          ; Initialize value at 0x40000000
    MOV R2,R1
    LDR R1,[R1]          ; Load no whose square is to be find
    MOV R1,R1,LSL#0x2    ; R1=Value at INPx4. to point at corresponding square
    ADD R0,R0,R1         ; Load address of element in Lookup table
    LDR R3, [R0]         ; Get square of given no in R3
    STR R3,[R2,#4]!
S   B   S
;Lookup table contains Squares of nos from 0 to 10
TABLE1  DCD 0,1,4,9,0x16,0x25,0x36,0x49,0x64,0x81,0x100;
    AREA DATA1,DATA,READWRITE
INP DCD 0X0045
    END                  ; Mark end of file
