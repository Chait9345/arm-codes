PINSEL1 EQU 0XE002C004; register to configure P0.16-P0.31 for specific operation
DACR    EQU 0XE006C000; DAC register address
    AREA CHANG, CODE,READONLY
    EXPORT __main  ;refer the line no 416 of startup.s file
__main
    ENTRY
        LDR R0,=PINSEL1;
        LDR R1,=DACR;
        LDR R2,=0X00080000; P0.25 as DAC output
        STR R2,[R0]
BACK    MOV R4, R3,LSL#6
        STR R4,[R1]
        ADD R3,R3,#1
        B BACK
        END
