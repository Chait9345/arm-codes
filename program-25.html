IO0DIR EQU 0xE0028008
IO0SET EQU 0xE0028004
IO0CLR EQU 0xE002800C
 AREA CHANG, CODE,READONLY
    EXPORT __main
__main
    ENTRY
        LDR r1,=IO0DIR ; load the address of the IODIR reg to R1
        LDR r0,=0xF0FF0000;  pins P0.16 to P0.23 and P0.28-P0.31 as output pins
        STR r0,[r1]         ;This configure P0.16 to P0.23 as output pins
        LDR r2,=IO0SET ; load the address of the IOCLR reg to R2
        LDR r3,=IO0CLR   ; load the address of th
S       STR R0,[R3]      ; Turn off display
        LDR R1,=0X803F0000 ; To display 0 on rightmost segment
        STR R1,[R2]
        BL DELAY
        STR R0,[R3]      ; Turn off display
        LDR R1,=0X40060000;To display 1 on third segment
        STR R1,[R2]
        BL DELAY
        STR R0,[R3]      ; Turn off display
        LDR R1,=0X205B0000;To display 2 on second segment
        STR R1,[R2]
        BL DELAY
        STR R0,[R3]      ; Turn off display
        LDR R1,=0X104F0000;To display 3 on leftmost segment
        STR R1,[R2]
        BL DELAY
        B   S
DELAY
        LDR r5, =0x2FF ; Delay program to retain the bit for some time.
delay   SUBS r5, r5, #1
        BNE delay
        BX LR
        END
