IO0DIR EQU 0xE0028008 ;Address for configuring Port0 as I/O
IO0SET EQU 0xE0028004 ;Register address for setting Port0 pins
IO0CLR EQU 0xE002800C ;Register address for clearing Port0 pins
 AREA CHANG, CODE,READONLY
    EXPORT __main  ;refer the line no 416 of startup.s file
__main
    ENTRY
        LDR r1,=IO0DIR ; load the address of the IODIR reg to R1
        LDR r0,=0xF000; To set P0.9 as output pin for buzzer
        STR r0,[r1]         ;This configure P0.12-P0.15 as output pins
        LDR r2,=IO0CLR ; load the address of the IOCLR reg to R2
        LDR r3,=IO0SET ; load the address of the IOSET reg to R3
S       LDR R5,=0X8000 ; Energise single coil at a time
BACK    STR R0,[R3]    ; Set all 4 pins to 1 to turnoff motor
        STR R5,[R2]    ; Clears one bit at a time to enable a coil(Active low)
        LDR R4,=0x2FFF; Keep varying this value for different speed
delay   SUBS R4,R4,#1;
        BNE delay
        MOV R5,R5,LSR #1; shift for energising next coil
        CMP R5,0X800    ; Are all coils energised?
        BNE BACK
        B S
        END
