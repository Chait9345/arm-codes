 INCLUDE VICVPB.S
    AREA CHANG, CODE,READONLY
    EXPORT __main  ;refer the line no 416 of startup.s file
__main
    ENTRY
        LDR r1,=IO0DIR ; load the address of the IODIR reg to R1
        LDR r0,=0xFF0000;  P0.0 to P0.7 as output
        STR r0,[r1]      ;This configure P0.0 to P0.7  as output pins
        LDR r2,=IO0CLR  ; load the address of the IOCLR reg to R2
        LDR r3,=IO0SET   ;
BACK    STR R0,[R2]
        BL DELAY
        STR R0,[R3]
        BL DELAY
        B BACK
DELAY   ;subroutine
        LDR R5,=0x2FFF
up      SUBS R5,R5,#1;
        BNE up
        BX LR
        END
