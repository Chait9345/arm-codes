IO0DIR EQU 0xE0028008
IO0SET EQU 0xE0028004
IO0CLR EQU 0xE002800C
 AREA CHANG, CODE,READONLY
    EXPORT __main  ;refer the line no 416 of startup.s file
__main
    ENTRY
        LDR r1,=IO0DIR ; load the address of the IODIR reg to R1
        LDR r0,=0x900; To set P0.8 AND P0.11 as output pins for the motor
        STR r0,[r1]         ;This configure P0.8and P0.11 as output pins
        LDR r2,=IO0CLR ; load the address of the IOCLR reg to R2
        LDR r3,=IO0SET ; load the address of the IOSET reg to R3
        ;LDR R5,=0X800 ; TO SWITCH DIRECTION
        LDR R6,=0X100 ; TO TURN ON OR OFF THE MOTOR
        STR R0,[R2] ; keep the motor off
BACK    STR R6,[R3]    ;TURN ON MOTOR ANTICLOCKWISE DIRECTION
        BL delay
        STR R0,[R2]    ;TURN OFF MOTOR
        BL delay
        STR R0,[R3]; Motor to rotate clockwise
        BL delay
        STR R0,[R2] ;TURN OFF MOTOR
        BL delay
        B BACK
delay   ;subroutine
        LDR R4,=0x2FFFFF
up      SUBS R4,R4,#1;
        BNE up
        BX LR
        END
