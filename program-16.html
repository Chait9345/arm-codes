; PROGRAM TO ADD an array of unsigned 16-BIT NUMBERS & STORE IN INTERNAL RAM
; THE RESULT CAN BE VIEWED IN LOCATION 0X40000000 & ALSO IN R0
    AREA  ADDITION , CODE, READONLY
ENTRY                  ;Mark first instruction to execute
START
    MOV R0, #0           ; INTIALISE SUM TO ZERO
    mov r5, #10          ; number of half words to add
    LDR R6, =num_array   ; points to the array
next_element
    LDRH R1, [r6]        ; read the number from array into r1

    ADD R0, R0, R1       ; ADD THE ELEMENTS
    add r6, r6, #2       ; update pointer for next element
    subs r5, r5, #1      ; reduce the count of number of elements in the array
    bne next_element

    LDR R4, =RESULT      ; LOADS THE ADDRESS OF RESULT
    STR R0, [R4]         ; STORES THE RESULT IN R1

here b here

num_array DCW   0X1111, 0X2222, 0x3333, 0x4444, 0x5555, 0x6666, 0XAAAA, 0XBBBB, 0xCCCC, 0XDDDD

    AREA DATA2, DATA, READWRITE     ; TO STORE RESULT IN GIVEN ADDRESS
RESULT DCD 0X0, 0x0
    END           ; Mark end of file
