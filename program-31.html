 INCLUDE VICVPB.S
 AREA T0_INT, CODE,READONLY
 EXPORT __main  ;refer the line no 416 of startup.s file
__main
    ENTRY
    LDR R1,=VPBDIV; For Pclk = 30MHz
    LDR R0,=0X02
    STR R0,[R1]
;   LDR R1,=MEMMAP; User Flash Mode. Interrupt vectors are
;   LDR R0,=0X01  ;not re-mapped and reside in Flash.
;   STR R0,[R1]
    LDR R1,=IO0DIR    ;
    LDR R0,=0x00FF0000;Configure P0.16-23 and P0.28,29 as output for seven segment
    STR R0,[R1]
    STR R0,[R1]
    LDR R1,=VICVectCntl0; slot0 with highest priority
    LDR R0,=0x24; T0 number is 4 and 2  for vectored IRQ slot is enabled
    STR R0,[R1]
    LDR R1,=VICVectAddr0
    LDR R0,=IRQ_TIM0; ISR address for Timer0 to be stored in VICVectAddr0
    STR R0,[R1]
    LDR R1,=VICIntEnable
    LDR R0,=0x10; enable Timer0
    STR R0,[R1]
    LDR R1,=T0PR;LOAD 29 SO THAT 30MHz/30= 1 microseconds
    LDR R0,=29;
    STR R0,[R1]
    LDR R1,=T0TCR
    LDR R0,=1;  Enable timer
    STR R0,[R1]
    LDR R1,=T0MR0;
    LDR R0,=1000000; Load timer match register0 to generate 1 second
    STR R0,[R1]
    LDR R1,=T0MCR;
    LDR R0,=3;  To generate interrupt on match Interrupt on MR0
    STR R0,[R1] ; and Reset on MR0: the TC will be reset if MR0 matches it.
    LDR R2,=0XFF; initial count 0
S   B S
IRQ_TIM0
    STMDB R13!,{R0-R1}
    LDR R1,=T0IR;
    LDR R0,=1
    STR R0,[R1]
    LDR R1,=VICVectAddr
    LDR R0,=0
    STR R0,[R1]
    LDR R1,=IO0PIN
    SUB R2,R2,#1
    MOV R0, R2,LSL#16
    STR R0,[R1]
    LDMIA R13!,{R0-R1}
    subs pc,r14,#4 ;Adjust the PC to jump back to where it was interrupted
    END
